// Animation d'un stickman en JS natif
const canvas = document.getElementById('stickmanCanvas');
const ctx = canvas.getContext('2d');

let x = 50;
function animate() {
    ctx.clearRect(0, 0, canvas.width, canvas.height);
    
    // Dessin du stickman en mouvement
    ctx.strokeStyle = '#4ec9b0';
    ctx.lineWidth = 2;
    
    // Tête
    ctx.beginPath();
    ctx.arc(x, 45, 10, 0, Math.PI * 2);
    ctx.stroke();

    x += 0.5; // Vitesse lente
    requestAnimationFrame(animate);
}
animate();
