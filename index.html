// Animation Alan Becker Style - Avancement & Mouvement dynamique
class Stickman {
    constructor(x, y) {
        this.x = x;
        this.y = y;
        this.speed = 0.35; // Vitesse lente d'avancement
        this.color = '#ff3355'; // Rouge Alan Becker
    }

    update() {
        this.x += this.speed;
        this.drawMotionLines(); // Effet de vitesse visible en capture
    }
}
